CREATE DATABASE IF NOT EXISTS

 -- 1. Source Table: Patient Intake

CREATE TABLE hospital_patient_intake (
    patient_id INT AUTO_INCREMENT PRIMARY KEY,
    patient_name VARCHAR(100),
    age INT,
    gender VARCHAR(10),
    disease VARCHAR(120),
    admission_type VARCHAR(20),
    condition_status VARCHAR(20),
    triage_score INT,
    doctor_name VARCHAR(100),
    admission_date DATE,
    mobile VARCHAR(15),
    transfer_status VARCHAR(20) DEFAULT 'PENDING',
    processed_at DATETIME DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 2. Destination Table: Critical Care

CREATE TABLE critical_care_patients (
    critical_id INT AUTO_INCREMENT PRIMARY KEY,
    source_patient_id INT UNIQUE,
    patient_name VARCHAR(100),
    age INT,
    disease VARCHAR(120),
    admission_type VARCHAR(20),
    condition_status VARCHAR(20),
    triage_score INT,
    doctor_name VARCHAR(100),
    routed_at DATETIME DEFAULT CURRENT_TIMESTAMP
);


-- 3. Destination Table: General Care

CREATE TABLE general_care_patients (
    general_id INT AUTO_INCREMENT PRIMARY KEY,
    source_patient_id INT UNIQUE,
    patient_name VARCHAR(100),
    age INT,
    disease VARCHAR(120),
    admission_type VARCHAR(20),
    condition_status VARCHAR(20),
    triage_score INT,
    doctor_name VARCHAR(100),
    routed_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Verify that all three tables exist
SHOW TABLES;
